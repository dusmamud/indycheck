indycheck validators are plain functions, so they drop into any form library. These patterns are the recommended ones; the snippets are complete and copy-pasteable.

The npm package also ships ready-made Zod schemas (`indycheck/zod`) — `panSchema`, `aadhaarSchema`, `mobileSchema`, and one per validator — if you would rather import a schema than write a `refine`.

## Zod (TypeScript)

```ts
import { z } from 'zod';
import { validatePan, validateAadhaar, validateMobileWithOptions } from 'indycheck';

// A Zod schema that reuses indycheck's bilingual reasons:
const panSchema = z
  .string()
  .refine((v) => validatePan(v).valid, (v) => ({
    message: validatePan(v).reason_en,
  }));

const signupSchema = z.object({
  pan: panSchema,
  mobile: z.string().refine((v) => validateMobileWithOptions(v, { strict: false }).valid, {
    message: 'Enter a valid 10-digit mobile number.',
  }),
  aadhaar: z.string().refine((v) => validateAadhaar(v).valid, {
    message: 'Enter a valid 12-digit Aadhaar number.',
  }),
});

// Normalise after parse: store the canonical value, not the raw input.
// (Only mobile results carry `parsed.normalized` — the 10-digit number.)
const parsed = signupSchema.parse({ pan: 'aaapl1234c', mobile: '+91 98765 43210', aadhaar: '2345 6789 0124' });
console.log(validateMobile(parsed.mobile).parsed.normalized); // '9876543210'
```

```ts
// Reusable helper: one schema factory per ID type.
import { validatePan, validateAadhaar, validateGstin, validateMobile, validatePincode } from 'indycheck';

const VALIDATORS = {
  pan: validatePan,
  aadhaar: validateAadhaar,
  gstin: validateGstin,
  mobile: validateMobile,
  pincode: validatePincode,
} as const;

type IndycheckType = keyof typeof VALIDATORS;

function indycheckSchema(type: IndycheckType) {
  const validate = VALIDATORS[type];
  return z.string().refine((v) => validate(v).valid, (v) => ({
    message: validate(v).reason_en,
  }));
}

const gstinSchema = indycheckSchema('gstin');
```

## React Hook Form

```tsx
import { useForm } from 'react-hook-form';
import { validatePan, validateMobileWithOptions } from 'indycheck';

type FormValues = { pan: string; mobile: string };

export function KycForm() {
  const {
    register,
    handleSubmit,
    formState: { errors },
  } = useForm<FormValues>();

  return (
    <form onSubmit={handleSubmit((data) => console.log(data))}>
      <input
        {...register('pan', {
          validate: (v) => {
            const r = validatePan(v);
            return r.valid || r.reason_en;
          },
        })}
        placeholder="AAAPL1234C"
      />
      {errors.pan && <p>{errors.pan.message}</p>}

      <input
        {...register('mobile', {
          validate: (v) => {
            const r = validateMobileWithOptions(v, { strict: false });
            return r.valid || r.reason_en;
          },
        })}
        placeholder="9876543210"
      />
      {errors.mobile && <p>{errors.mobile.message}</p>}

      <button type="submit">Submit</button>
    </form>
  );
}
```

Use `strict: false` in forms — users paste values with spaces and lowercase letters. Show `reason_hi` instead of `reason_en` when the user's language is Hindi.

## Yup

```js
import * as yup from 'yup';
import { validateGstin, validatePincode } from 'indycheck';

const vendorSchema = yup.object({
  gstin: yup
    .string()
    .required('GSTIN is required.')
    .test('gstin', (value) => {
      const r = validateGstin(value ?? '');
      return r.valid || r.reason_en;
    }),
  pincode: yup
    .string()
    .required('PIN code is required.')
    .test('pincode', (value) => {
      const r = validatePincode(value ?? '');
      return r.valid || r.reason_en;
    }),
});
```

## Pydantic (Python)

```python
from pydantic import BaseModel, field_validator
from indycheck import (
    format_gstin,
    format_pan,
    validate_gstin,
    validate_mobile,
    validate_pan,
)

class Vendor(BaseModel):
    pan: str
    mobile: str
    gstin: str | None = None

    @field_validator("pan")
    @classmethod
    def check_pan(cls, v: str) -> str:
        r = validate_pan(v)
        if not r["valid"]:
            raise ValueError(r["reason_en"])
        return format_pan(v)  # 'AAAPL1234C' — the canonical value

    @field_validator("mobile")
    @classmethod
    def check_mobile(cls, v: str) -> str:
        r = validate_mobile(v)
        if not r["valid"]:
            raise ValueError(r["reason_en"])
        return r["parsed"]["normalized"]  # '9876543210'

    @field_validator("gstin")
    @classmethod
    def check_gstin(cls, v: str | None) -> str | None:
        if v is None:
            return None
        r = validate_gstin(v)
        if not r["valid"]:
            raise ValueError(r["reason_en"])
        return format_gstin(v)  # '27AAAPL1234C1ZE'
```

```python
# The parsed breakdown is available for business logic:
from indycheck import validate_gstin

r = validate_gstin("27AAAPL1234C1ZE")
if r["valid"] and r["parsed"]["state"] != "Maharashtra":
    print("Vendor is outside Maharashtra — different tax handling.")
```

## FastAPI (Python)

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from indycheck import (
    validate_aadhaar,
    validate_gstin,
    validate_mobile,
    validate_pan,
    validate_pincode,
)

app = FastAPI()

VALIDATORS = {
    "pan": validate_pan,
    "aadhaar": validate_aadhaar,
    "gstin": validate_gstin,
    "mobile": validate_mobile,
    "pincode": validate_pincode,
}

class CheckRequest(BaseModel):
    type: str
    value: str

@app.post("/check")
def check(req: CheckRequest):
    validate = VALIDATORS.get(req.type)
    if validate is None:
        raise HTTPException(status_code=422, detail={"code": "UNKNOWN_TYPE"})
    r = validate(req.value)
    if not r["valid"]:
        raise HTTPException(status_code=422, detail={
            "code": r["code"],
            "reason_en": r["reason_en"],
            "reason_hi": r["reason_hi"],
        })
    return {"valid": True, "parsed": r["parsed"]}
```

<Aside type="caution" title="Still format-only">
  These integrations make your forms reject mistyped IDs instantly — they do not verify that an ID is real. Keep the real verification step (OTP, authorised APIs) behind the form.
</Aside>
